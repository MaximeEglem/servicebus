# ServiceBus CLI (`sb`) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `sb`, a cross-platform Go CLI and library for managing Azure Service Bus queues, topics, subscriptions, and messages — with rich dead letter management, multi-environment contexts, and MCP/FFI integration.

**Architecture:** `pkg/` is the clean public library (management, messaging, auth, ffi) with zero dependency on CLI concerns. `cmd/sb/` is the Cobra CLI — one thin consumer of `pkg/`. `internal/` packages (config, format) are shared by CLI commands only. Tests use interfaces and mocks throughout to avoid requiring a live Azure connection.

**Tech Stack:** Go 1.22+, `github.com/spf13/cobra` v1.8, `github.com/Azure/azure-sdk-for-go/sdk/messaging/azservicebus` v1.7, `github.com/Azure/azure-sdk-for-go/sdk/azidentity` v1.7, `github.com/olekukonko/tablewriter` v0.0.5, `github.com/fatih/color` v1.16, `gopkg.in/yaml.v3` v3.0.1

```go
package config

import (
	"os"

	"gopkg.in/yaml.v3"
)

type Config struct {
	CurrentContext string             `yaml:"current-context"`
	Contexts       map[string]Context `yaml:"contexts"`
	Settings       Settings           `yaml:"settings"`
}

type Context struct {
	ConnectionString      string `yaml:"connection-string,omitempty"`
	Namespace             string `yaml:"namespace,omitempty"`
	UseDefaultCredentials bool   `yaml:"use-default-credentials,omitempty"`
}

type Settings struct {
	MaxItems            int        `yaml:"max-items"`
	WorkingDirectory    string     `yaml:"working-directory"`
	CacheExpirationDays int        `yaml:"cache-expiration-days"`
	Thresholds          Thresholds `yaml:"thresholds"`
}

type Thresholds struct {
	SizePercentage int `yaml:"size-percentage"`
	LastAccessDays int `yaml:"last-access-days"`
}

func defaults() *Config {
	return &Config{
		Contexts: map[string]Context{},
		Settings: Settings{
			MaxItems:            50,
			WorkingDirectory:    "~/.sb/exports",
			CacheExpirationDays: 30,
			Thresholds:          Thresholds{SizePercentage: 70, LastAccessDays: 30},
		},
	}
}

func Load() (*Config, error) {
	data, err := os.ReadFile(ConfigPath())
	if os.IsNotExist(err) {
		return defaults(), nil
	}
	if err != nil {
		return nil, err
	}
	cfg := defaults()
	if err := yaml.Unmarshal(data, cfg); err != nil {
		return nil, err
	}
	if cfg.Contexts == nil {
		cfg.Contexts = map[string]Context{}
	}
	return cfg, nil
}

func Save(cfg *Config) error {
	if err := os.MkdirAll(ConfigDir(), 0700); err != nil {
		return err
	}
	data, err := yaml.Marshal(cfg)
	if err != nil {
		return err
	}
	return os.WriteFile(ConfigPath(), data, 0600)
}
```

```go
package auth

import (
	"errors"
	"os"

	"github.com/Azure/azure-sdk-for-go/sdk/azcore"
	"github.com/Azure/azure-sdk-for-go/sdk/azidentity"
	"github.com/yourusername/sb/internal/config"
)

type Options struct {
	ConnectionString string
	Namespace        string
	ContextName      string
	Config           *config.Config
}

type Credentials struct {
	ConnectionString string
	Namespace        string
	Token            azcore.TokenCredential
}

// Resolve picks credentials in priority order:
// 1. explicit --connection-string flag
// 2. AZURE_SERVICEBUS_CONNECTION_STRING env var
// 3. named context from Config
// 4. --namespace flag → DefaultAzureCredential
func Resolve(opts Options) (*Credentials, error) {
	if opts.ConnectionString != "" {
		return &Credentials{ConnectionString: opts.ConnectionString}, nil
	}
	if cs := os.Getenv("AZURE_SERVICEBUS_CONNECTION_STRING"); cs != "" {
		return &Credentials{ConnectionString: cs}, nil
	}
	if opts.ContextName != "" {
		if opts.Config == nil {
			return nil, errors.New("context specified but no config loaded")
		}
		ctx, ok := opts.Config.Contexts[opts.ContextName]
		if !ok {
			return nil, errors.New("context not found: " + opts.ContextName)
		}
		return fromContext(ctx)
	}
	if opts.Namespace != "" {
		return azureAD(opts.Namespace)
	}
	return nil, errors.New("no credentials: use --connection-string, set AZURE_SERVICEBUS_CONNECTION_STRING, or run 'sb config set-context'")
}

func fromContext(ctx config.Context) (*Credentials, error) {
	if ctx.ConnectionString != "" {
		return &Credentials{ConnectionString: ctx.ConnectionString}, nil
	}
	return azureAD(ctx.Namespace)
}

func azureAD(namespace string) (*Credentials, error) {
	cred, err := azidentity.NewDefaultAzureCredential(nil)
	if err != nil {
		return nil, err
	}
	return &Credentials{Namespace: namespace, Token: cred}, nil
}
```

```go
package format

import (
	"encoding/json"
	"fmt"
	"io"
	"strings"

	"gopkg.in/yaml.v3"
)

type OutputFormat string

const (
	Table OutputFormat = "table"
	JSON  OutputFormat = "json"
	YAML  OutputFormat = "yaml"
)

func ParseFormat(s string) (OutputFormat, error) {
	switch strings.ToLower(s) {
	case "table", "":
		return Table, nil
	case "json":
		return JSON, nil
	case "yaml":
		return YAML, nil
	default:
		return "", fmt.Errorf("unknown output format %q: use table, json, or yaml", s)
	}
}

type ThresholdConfig struct {
	SizePercentage int // highlight rows where SizePercentage >= this value (red)
	LastAccessDays int // highlight rows where LastAccessDays >= this value (yellow)
}

type Printer struct {
	format     OutputFormat
	out        io.Writer
	thresholds ThresholdConfig
}

func NewPrinter(f OutputFormat, out io.Writer, t ThresholdConfig) *Printer {
	return &Printer{format: f, out: out, thresholds: t}
}

func (p *Printer) Print(data any) error {
	switch p.format {
	case JSON:
		return printJSON(p.out, data)
	case YAML:
		return printYAML(p.out, data)
	default:
		return printTable(p.out, data, p.thresholds)
	}
}

func printJSON(w io.Writer, data any) error {
	enc := json.NewEncoder(w)
	enc.SetIndent("", "  ")
	return enc.Encode(data)
}

func printYAML(w io.Writer, data any) error {
	return yaml.NewEncoder(w).Encode(data)
}
```

- [ ] **Step 4: Create `internal/format/table.go`**

```go
package format

import (
	"encoding/json"
	"io"
	"reflect"
	"strconv"
	"time"

	"github.com/fatih/color"
	"github.com/olekukonko/tablewriter"
)

var (
	red    = color.New(color.FgRed).SprintFunc()
	yellow = color.New(color.FgYellow).SprintFunc()
)

// printTable renders any slice of structs as a table.
// Struct fields are column headers; values are stringified.
// Rows implement Highlighter to apply threshold colours.
type Highlighter interface {
	// HighlightColor returns "red", "yellow", or "" based on thresholds.
	HighlightColor(t ThresholdConfig) string
}

func printTable(w io.Writer, data any, t ThresholdConfig) error {
	// Convert data to []map[string]string via JSON round-trip for simplicity.
	b, err := json.Marshal(data)
	if err != nil {
		return err
	}
	var rows []map[string]any
	if err := json.Unmarshal(b, &rows); err != nil {
		return err
	}
	if len(rows) == 0 {
		return nil
	}

	// Determine column order from struct field names.
	headers := columnHeaders(data)
	table := tablewriter.NewWriter(w)
	table.SetHeader(headers)
	table.SetBorder(false)
	table.SetHeaderAlignment(tablewriter.ALIGN_LEFT)
	table.SetAlignment(tablewriter.ALIGN_LEFT)
	table.SetColumnSeparator("  ")

	// Apply highlight colour from Highlighter interface if the slice element implements it.
	hlColors := highlightColors(data, t)

	for i, row := range rows {
		cells := make([]string, len(headers))
		for j, h := range headers {
			cells[j] = stringify(row[jsonKey(h)])
		}
		if i < len(hlColors) && hlColors[i] != "" {
			coloured := make([]string, len(cells))
			for j, c := range cells {
				switch hlColors[i] {
				case "red":
					coloured[j] = red(c)
				case "yellow":
					coloured[j] = yellow(c)
				default:
					coloured[j] = c
				}
			}
			table.Append(coloured)
		} else {
			table.Append(cells)
		}
	}
	table.Render()
	return nil
}

func columnHeaders(data any) []string {

func stringify(v any) string {

	case float64:

	case bool:

			return "true"
		}
		return "false"
	case time.Time:

	default:
		b, _ := json.Marshal(v)
```

```go
import (
	"fmt"
	"regexp"
	"strconv"
	"strings"
	"time"
)

type FilterKind int

const (
	FilterKindKeyword       FilterKind = iota // plain text match on name
	FilterKindNumericField                    // Field:min-max
	FilterKindDateField                       // Field:dd/mm/yyyy-dd/mm/yyyy
	FilterKindErrorField                      // ErrorType/ErrorMessage/ErrorDate:value
	FilterKindDLQDate                         // dd/mm/yyyy (enqueued date on DL message)
	FilterKindSequenceRange                   // 100-200 or single 200
)

type Filter struct {
	Kind     FilterKind
	Keyword  string
	Field    string
	MinValue float64
	MaxValue float64
	MinDate  time.Time
	MaxDate  time.Time
	SeqMin   int64
	SeqMax   int64
}

var (
	reDateRange  = regexp.MustCompile(`^(\d{2}/\d{2}/\d{4})-(\d{2}/\d{2}/\d{4})$`)
	reSingleDate = regexp.MustCompile(`^\d{2}/\d{2}/\d{4}$`)
	reNumRange   = regexp.MustCompile(`^(\d+)-(\d+)$`)
	reSingleNum  = regexp.MustCompile(`^\d+$`)
	dateLayout   = "02/01/2006"
	errorFields  = map[string]bool{"ErrorType": true, "ErrorMessage": true, "ErrorDate": true}
	numericFields = map[string]bool{
		"ActiveMessages": true, "DeadLetterMessages": true, "SizeInMegabytes": true,
	}
	dateFields = map[string]bool{"CreationDate": true, "LastAccessDate": true, "ErrorDate": true}
)

// ParseFilter parses a single -f value into a Filter.
func ParseFilter(s string) (Filter, error) {
	if idx := strings.Index(s, ":"); idx > 0 {
		field := s[:idx]
		value := s[idx+1:]
		return parseFieldFilter(field, value)
	}
	// No colon: could be date, sequence range, or keyword.
	if reSingleDate.MatchString(s) {
		d, err := time.Parse(dateLayout, s)
		if err != nil {
			return Filter{}, err
		}
		return Filter{Kind: FilterKindDLQDate, MinDate: d, MaxDate: d.Add(24 * time.Hour)}, nil
	}
	if m := reNumRange.FindStringSubmatch(s); m != nil {
		min, _ := strconv.ParseInt(m[1], 10, 64)
		max, _ := strconv.ParseInt(m[2], 10, 64)
		return Filter{Kind: FilterKindSequenceRange, SeqMin: min, SeqMax: max}, nil
	}
	if reSingleNum.MatchString(s) {
		n, _ := strconv.ParseInt(s, 10, 64)
		return Filter{Kind: FilterKindSequenceRange, SeqMin: n, SeqMax: n}, nil
	}
	return Filter{Kind: FilterKindKeyword, Keyword: strings.ToLower(s)}, nil
}

func parseFieldFilter(field, value string) (Filter, error) {
	if errorFields[field] {
		if field == "ErrorDate" {
			return parseDateRangeFilter(field, value)
		}
		return Filter{Kind: FilterKindErrorField, Field: field, Keyword: value}, nil
	}
	if numericFields[field] {
		return parseNumericFilter(field, value)
	}
	if dateFields[field] {
		return parseDateRangeFilter(field, value)
	}
	// Unknown field — treat as keyword with field prefix for extensibility.
	return Filter{Kind: FilterKindKeyword, Keyword: strings.ToLower(field + ":" + value)}, nil
}

func parseNumericFilter(field, value string) (Filter, error) {
	if m := reNumRange.FindStringSubmatch(value); m != nil {
		min, _ := strconv.ParseFloat(m[1], 64)
		max, _ := strconv.ParseFloat(m[2], 64)
		return Filter{Kind: FilterKindNumericField, Field: field, MinValue: min, MaxValue: max}, nil
	}
	n, err := strconv.ParseFloat(value, 64)
	if err != nil {
		return Filter{}, fmt.Errorf("invalid numeric filter %q: %w", value, err)
	}
	return Filter{Kind: FilterKindNumericField, Field: field, MinValue: n, MaxValue: n}, nil
}

func parseDateRangeFilter(field, value string) (Filter, error) {
	if m := reDateRange.FindStringSubmatch(value); m != nil {
		min, err := time.Parse(dateLayout, m[1])
		if err != nil {
			return Filter{}, err
		}
		max, err := time.Parse(dateLayout, m[2])
		if err != nil {
			return Filter{}, err
		}
		return Filter{Kind: FilterKindDateField, Field: field, MinDate: min, MaxDate: max.Add(24 * time.Hour)}, nil
	}
	d, err := time.Parse(dateLayout, value)
	if err != nil {
		return Filter{}, fmt.Errorf("invalid date %q: expected dd/mm/yyyy", value)
	}
	return Filter{Kind: FilterKindDateField, Field: field, MinDate: d, MaxDate: d.Add(24 * time.Hour)}, nil
}

		return queues
```

```go
package management

// QueueCompareResult is one row in the N-way queue comparison.
type QueueCompareResult struct {
	Name      string
	PresentIn map[string]bool   // context name → present
	Counts    map[string]int64  // context name → ActiveMessages
	DLCounts  map[string]int64  // context name → DeadLetterMessages
	HasIssue  bool              // true if missing in any context or counts diverge significantly
}

// CompareQueues performs an N-way diff of queue lists across contexts.
// contexts maps context name → queue list for that context.
func CompareQueues(contexts map[string][]QueueInfo) []QueueCompareResult {
	// Collect all unique queue names.
	names := map[string]bool{}
	for _, queues := range contexts {
		for _, q := range queues {
			names[q.Name] = true
		}
	}

	var results []QueueCompareResult
	for name := range names {
		r := QueueCompareResult{
			Name:      name,
			PresentIn: map[string]bool{},
			Counts:    map[string]int64{},
			DLCounts:  map[string]int64{},
		}
		for ctx, queues := range contexts {
			for _, q := range queues {
				if q.Name == name {
					r.PresentIn[ctx] = true
					r.Counts[ctx] = q.ActiveMessages
					r.DLCounts[ctx] = q.DeadLetterMessages
				}
			}
		}
		// HasIssue: missing in any context, or active messages differ by >10x between any two.
		if len(r.PresentIn) < len(contexts) {
			r.HasIssue = true
		} else {
			var min, max int64 = 1<<62, 0
			for _, c := range r.Counts {
				if c < min {
					min = c
				}
				if c > max {
					max = c
				}
			}
			if max > 0 && min == 0 {
				r.HasIssue = true
			} else if min > 0 && max/min > 10 {
				r.HasIssue = true
			}
		}
		results = append(results, r)
	}
	return results
}

// CompareTopics performs the same N-way diff for topics.
type TopicCompareResult struct {
	Name      string
	PresentIn map[string]bool
	Sizes     map[string]int64
	HasIssue  bool
}

func CompareTopics(contexts map[string][]TopicInfo) []TopicCompareResult {
	names := map[string]bool{}
	for _, topics := range contexts {
		for _, t := range topics {
			names[t.Name] = true
		}
	}
	var results []TopicCompareResult
	for name := range names {
		r := TopicCompareResult{
			Name:      name,
			PresentIn: map[string]bool{},
			Sizes:     map[string]int64{},
		}
		for ctx, topics := range contexts {
			for _, t := range topics {
				if t.Name == name {
					r.PresentIn[ctx] = true
					r.Sizes[ctx] = t.SizeInMegabytes
				}
			}
		}
		if len(r.PresentIn) < len(contexts) {
			r.HasIssue = true
		}
		results = append(results, r)
	}
	return results
}
```

```go
	}
	if err != nil {
		return nil, err
	}
	s, err := client.NewSender(queueOrTopic, nil)
	if err != nil {
		return nil, err
	}
	return &azureSender{sender: s}, nil
}

func (a *azureSender) Send(ctx context.Context, opts SendOptions) error {
	msg := &azservicebus.Message{Body: opts.Body}
	if opts.Properties != nil {
		msg.ApplicationProperties = opts.Properties
	}
	if opts.DelayMins > 0 {
		at := time.Now().Add(time.Duration(opts.DelayMins) * time.Minute)
		_, err := a.sender.ScheduleMessages(ctx, []*azservicebus.Message{msg}, at, nil)
		return err
	}
	return a.sender.SendMessage(ctx, msg, nil)
}

// SendWithSender sends opts.Body opts.Count times (minimum 1).
func SendWithSender(ctx context.Context, s Sender, opts SendOptions) error {
	count := opts.Count
	if count < 1 {
		count = 1
	}
	for i := 0; i < count; i++ {
		if err := s.Send(ctx, opts); err != nil {
			return err
		}
	}
	return nil
}

// SendFromPath sends from a JSON file or all *.json files in a directory.
func SendFromPath(ctx context.Context, s Sender, path string, base SendOptions) error {
	info, err := os.Stat(path)
	if err != nil {
		return err
	}
	if !info.IsDir() {
		return sendFile(ctx, s, path, base)
	}
	entries, err := os.ReadDir(path)
	if err != nil {
		return err
	}
	for _, e := range entries {
		if e.IsDir() || !strings.HasSuffix(e.Name(), ".json") {
			continue
		}
		if err := sendFile(ctx, s, filepath.Join(path, e.Name()), base); err != nil {
			return err
		}
	}
	return nil
}

func sendFile(ctx context.Context, s Sender, path string, base SendOptions) error {
	data, err := os.ReadFile(path)
	if err != nil {
		return err
	}
	// Parse to extract properties if the JSON has a top-level "__properties" key.
	var raw map[string]json.RawMessage
	var props map[string]interface{}
	if err := json.Unmarshal(data, &raw); err == nil {
		if p, ok := raw["__properties"]; ok {
			json.Unmarshal(p, &props)
			delete(raw, "__properties")
			data, _ = json.Marshal(raw)
		}
	}
	opts := base
	opts.Body = data
	if props != nil {
		opts.Properties = props
	}
	return SendWithSender(ctx, s, opts)
}
```

```go
package messaging

import (
	"strings"

	"github.com/yourusername/sb/pkg/management"

// ApplyMessageFilters filters DL messages using the same Filter type as management.
// All filters must match (AND), multiple keywords are OR within themselves.
func ApplyMessageFilters(msgs []Message, filters []management.Filter) []Message {
	if len(filters) == 0 {
		return msgs
	}
	var result []Message
	for _, m := range msgs {
		if msgMatchesAll(m, filters) {
			result = append(result, m)
		}
	}
	return result
}

func msgMatchesAll(m Message, filters []management.Filter) bool {
	for _, f := range filters {
		if !msgMatchesFilter(m, f) {
			return false
		}
	}
	return true
}

func msgMatchesFilter(m Message, f management.Filter) bool {
	switch f.Kind {
	case management.FilterKindKeyword:
		return strings.Contains(strings.ToLower(m.Body), f.Keyword) ||
			strings.Contains(strings.ToLower(m.DeadLetterReason), f.Keyword) ||
			strings.Contains(strings.ToLower(m.DeadLetterErrorDescription), f.Keyword)
	case management.FilterKindSequenceRange:
		return m.SequenceNumber >= f.SeqMin && m.SequenceNumber <= f.SeqMax
	case management.FilterKindDLQDate:
		return !m.EnqueuedAt.Before(f.MinDate) && m.EnqueuedAt.Before(f.MaxDate)
	case management.FilterKindDateField:
		return !m.EnqueuedAt.Before(f.MinDate) && m.EnqueuedAt.Before(f.MaxDate)
	case management.FilterKindErrorField:
		switch f.Field {
		case "ErrorType":
			return strings.EqualFold(m.DeadLetterReason, f.Keyword)
		case "ErrorMessage":
			return strings.Contains(strings.ToLower(m.DeadLetterErrorDescription), strings.ToLower(f.Keyword))
		}
	}
	return true
}
```

```go
// DLReceiver manages dead letter sub-queues.
type DLReceiver interface {
	// Peek returns up to max messages from the DLQ without locking them.
	Peek(ctx context.Context, max int) ([]Message, error)
	// Purge receives and completes all messages matching filters.
	Purge(ctx context.Context, filters []management.Filter, max int) (int, error)
	// Replay receives DL messages matching filters and resends them to the main queue.
	Replay(ctx context.Context, filters []management.Filter, opts DLOptions) (int, error)
	// Export peeks all matching messages and writes them as JSON to outPath.
	Export(ctx context.Context, filters []management.Filter, max int, outPath string) error
}

type azureDLReceiver struct {
	receiver *azservicebus.Receiver
	sender   *azservicebus.Sender
}

// NewDLReceiverForQueue creates a DLReceiver for a queue's dead letter sub-queue.
func NewDLReceiverForQueue(cred *auth.Credentials, queueName string) (DLReceiver, error) {
	client, err := newClient(cred)
	if err != nil {
		return nil, err
	}
	recv, err := client.NewReceiverForQueue(queueName, &azservicebus.ReceiverOptions{
		SubQueue: azservicebus.SubQueueDeadLetter,
	})
	if err != nil {
		return nil, err
	}
	send, err := client.NewSender(queueName, nil)
	if err != nil {
		return nil, err
	}
	return &azureDLReceiver{receiver: recv, sender: send}, nil
}

// NewDLReceiverForSubscription creates a DLReceiver for a subscription's dead letter sub-queue.
func NewDLReceiverForSubscription(cred *auth.Credentials, topic, sub string) (DLReceiver, error) {
	client, err := newClient(cred)
	if err != nil {
		return nil, err
	}
	recv, err := client.NewReceiverForSubscription(topic, sub, &azservicebus.ReceiverOptions{
		SubQueue: azservicebus.SubQueueDeadLetter,
	})
	if err != nil {
		return nil, err
	}
	// For subscriptions, replay sends to the topic.
	send, err := client.NewSender(topic, nil)
	if err != nil {
		return nil, err
	}
	return &azureDLReceiver{receiver: recv, sender: send}, nil
}

func newClient(cred *auth.Credentials) (*azservicebus.Client, error) {
	if cred.ConnectionString != "" {
		return azservicebus.NewClientFromConnectionString(cred.ConnectionString, nil)
	}
	return azservicebus.NewClient(cred.Namespace, cred.Token, nil)
}

func (r *azureDLReceiver) Peek(ctx context.Context, max int) ([]Message, error) {
	if max <= 0 {
		max = 1000
	}
	peeked, err := r.receiver.PeekMessages(ctx, max, nil)
	if err != nil {
		return nil, err
	}
	return convertMessages(peeked), nil
}

func (r *azureDLReceiver) Purge(ctx context.Context, filters []management.Filter, max int) (int, error) {
	if max <= 0 {
		max = 1000
	}
	msgs, err := r.receiver.ReceiveMessages(ctx, max, nil)
	if err != nil {
		return 0, err
	}
	converted := convertReceivedMessages(msgs)
	matched := ApplyMessageFilters(converted, filters)
	// Build a lookup of sequence numbers to complete.
	seqSet := map[int64]bool{}
	for _, m := range matched {
		seqSet[m.SequenceNumber] = true
	}
	count := 0
	for i, m := range msgs {
		if seqSet[*converted[i].SequenceNumberPtr()] {
			if err := r.receiver.CompleteMessage(ctx, m, nil); err != nil {
				return count, fmt.Errorf("completing seq %d: %w", *m.SequenceNumber, err)
			}
			count++
		} else {
			r.receiver.AbandonMessage(ctx, m, nil)
		}
	}
	return count, nil
}

func (r *azureDLReceiver) Replay(ctx context.Context, filters []management.Filter, opts DLOptions) (int, error) {
	max := opts.Max
	if max <= 0 {
		max = 1000
	}
	msgs, err := r.receiver.ReceiveMessages(ctx, max, nil)
	if err != nil {
		return 0, err
	}
	converted := convertReceivedMessages(msgs)
	matched := ApplyMessageFilters(converted, filters)
	seqSet := map[int64]bool{}
	for _, m := range matched {
		seqSet[m.SequenceNumber] = true
	}

	count := 0
	for i, m := range msgs {
		seq := int64(0)
		if m.SequenceNumber != nil {
			seq = *m.SequenceNumber
		}
		if !seqSet[seq] {
			r.receiver.AbandonMessage(ctx, m, nil)
			continue
		}
		newMsg := &azservicebus.Message{
			Body:                  m.Body,
			ApplicationProperties: m.ApplicationProperties,
		}
		if opts.TTLDays > 0 {
			ttl := time.Duration(opts.TTLDays) * 24 * time.Hour
			newMsg.TimeToLive = &ttl
		}
		var sendErr error
		if opts.DelayMins > 0 {
			at := time.Now().Add(time.Duration(opts.DelayMins) * time.Minute)
			_, sendErr = r.sender.ScheduleMessages(ctx, []*azservicebus.Message{newMsg}, at, nil)
		} else {
			sendErr = r.sender.SendMessage(ctx, newMsg, nil)
		}
		if sendErr != nil {
			r.receiver.AbandonMessage(ctx, m, nil)
			return count, sendErr
		}
		if err := r.receiver.CompleteMessage(ctx, m, nil); err != nil {
			return count, err
		}
		count++
		_ = converted[i] // suppress unused warning
	}
	return count, nil
}

func (r *azureDLReceiver) Export(ctx context.Context, filters []management.Filter, max int, outPath string) error {
	msgs, err := r.Peek(ctx, max)
	if err != nil {
		return err
	}
	msgs = ApplyMessageFilters(msgs, filters)
	data, err := json.MarshalIndent(msgs, "", "  ")
	if err != nil {
		return err
	}
	if err := os.MkdirAll(filepath.Dir(outPath), 0755); err != nil {
		return err
	}
	return os.WriteFile(outPath, data, 0644)
}

func convertMessages(msgs []*azservicebus.ReceivedMessage) []Message {
	result := make([]Message, len(msgs))
	for i, m := range msgs {
		result[i] = toMessage(m)
	}
	return result
}

func convertReceivedMessages(msgs []*azservicebus.ReceivedMessage) []Message {
	return convertMessages(msgs)
}

func toMessage(m *azservicebus.ReceivedMessage) Message {
	msg := Message{
		Body:       string(m.Body),
		Properties: m.ApplicationProperties,
	}
	if m.MessageID != "" {
		msg.MessageID = m.MessageID
	}
	if m.SequenceNumber != nil {
		msg.SequenceNumber = *m.SequenceNumber
	}
	if m.EnqueuedTime != nil {
		msg.EnqueuedAt = *m.EnqueuedTime
	}
	if m.DeadLetterReason != nil {
		msg.DeadLetterReason = *m.DeadLetterReason
	}
	if m.DeadLetterErrorDescription != nil {
		msg.DeadLetterErrorDescription = *m.DeadLetterErrorDescription
	}
	return msg
}

// SequenceNumberPtr is a helper for Purge/Replay to avoid nil-deref.
func (m *Message) SequenceNumberPtr() *int64 { return &m.SequenceNumber }
```

```go
func resolveCredWithConfig(cfg *config.Config) (*auth.Credentials, error) {
	ctxName := gf.context
	if ctxName == "" {
		ctxName = cfg.CurrentContext
	}
	// Try session state context if still empty.
	if ctxName == "" {
		if state, err := config.LoadState(); err == nil {
			ctxName = state.Context
		}
	}
	return auth.Resolve(auth.Options{
		ConnectionString: gf.connectionString,
		Namespace:        gf.namespace,
		ContextName:      ctxName,
		Config:           cfg,
	})
}

func parseFilters(raw []string) ([]management.Filter, error) {
	var filters []management.Filter
	for _, s := range raw {
		f, err := management.ParseFilter(s)
		if err != nil {
			return nil, fmt.Errorf("invalid filter %q: %w", s, err)
		}
		filters = append(filters, f)
	}
	return filters, nil
}

func parseSortFlag(s string) (field, dir string) {
	for i := len(s) - 1; i >= 0; i-- {
		if s[i] == ':' {
			return s[:i], s[i+1:]
		}
	}
	return s, "asc"
}

func applyMax(queues []management.QueueInfo, cfg *config.Config) []management.QueueInfo {

		return queues[:limit]
	}
	return queues
}
```

```go
type mcpRequest struct {
	JSONRPC string          `json:"jsonrpc"`
	ID      any             `json:"id"`
	Method  string          `json:"method"`
	Params  json.RawMessage `json:"params"`
}

type mcpResponse struct {
	JSONRPC string `json:"jsonrpc"`
	ID      any    `json:"id"`
	Result  any    `json:"result,omitempty"`
	Error   *mcpError `json:"error,omitempty"`
}

type mcpError struct {
	Code    int    `json:"code"`
	Message string `json:"message"`
}

type mcpTool struct {
	Name        string         `json:"name"`
	Description string         `json:"description"`
	InputSchema map[string]any `json:"inputSchema"`
}

func buildMCPCmd() *cobra.Command {
	return &cobra.Command{
		Use:   "mcp",
		Short: "Start MCP stdio server (tools auto-generated from sb schema)",
		RunE:  runMCP,
	}
}

func runMCP(_ *cobra.Command, _ []string) error {
	tools := mcpToolsFromSchema()
	scanner := bufio.NewScanner(os.Stdin)
	enc := json.NewEncoder(os.Stdout)

	for scanner.Scan() {
		line := scanner.Text()
		if strings.TrimSpace(line) == "" {
			continue
		}
		var req mcpRequest
		if err := json.Unmarshal([]byte(line), &req); err != nil {
			continue
		}
		resp := handleMCPRequest(req, tools)
		enc.Encode(resp)
	}
	return scanner.Err()
}

func handleMCPRequest(req mcpRequest, tools []mcpTool) mcpResponse {
	base := mcpResponse{JSONRPC: "2.0", ID: req.ID}
	switch req.Method {
	case "initialize":
		base.Result = map[string]any{
			"protocolVersion": "2024-11-05",
			"capabilities":    map[string]any{"tools": map[string]any{}},
			"serverInfo":      map[string]any{"name": "sb", "version": "1.0.0"},
		}
	case "tools/list":
		base.Result = map[string]any{"tools": tools}
	case "tools/call":
		var params struct {
			Name      string         `json:"name"`
			Arguments map[string]any `json:"arguments"`
		}
		if err := json.Unmarshal(req.Params, &params); err != nil {
			base.Error = &mcpError{Code: -32600, Message: err.Error()}
			return base
		}
		out, err := invokeSBCommand(params.Name, params.Arguments)
		if err != nil {
			base.Error = &mcpError{Code: -32000, Message: err.Error()}
			return base
		}
		base.Result = map[string]any{
			"content": []map[string]any{{"type": "text", "text": out}},
		}
	default:
		base.Error = &mcpError{Code: -32601, Message: "method not found: " + req.Method}
	}
	return base
}

// invokeSBCommand re-invokes the sb binary with the given command and arguments.
func invokeSBCommand(toolName string, args map[string]any) (string, error) {
	// toolName is like "q_list", "dlq_view", etc. — convert to sb args.
	cmdArgs := mcpToolToArgs(toolName, args)
	self, err := os.Executable()
	if err != nil {
		self = "sb"
	}
	cmd := exec.Command(self, append(cmdArgs, "--output", "json")...)
	out, err := cmd.CombinedOutput()
	if err != nil {
		return "", fmt.Errorf("%s: %s", err, string(out))
	}
	return string(out), nil
}

func mcpToolToArgs(toolName string, args map[string]any) []string {
	// toolName format: "<command>_<subcommand>" e.g. "q_list", "dlq_view"
	parts := strings.SplitN(toolName, "_", 2)
	result := []string{parts[0]}
	if len(parts) == 2 && parts[1] != "list" {
		result = append(result, parts[1])
	}
	for k, v := range args {
		switch k {
		case "context":
			result = append(result, "-c", fmt.Sprintf("%v", v))
		case "name":
			result = append(result, "-n", fmt.Sprintf("%v", v))
		case "topic":
			result = append(result, "-t", fmt.Sprintf("%v", v))
		case "filter":
			result = append(result, "-f", fmt.Sprintf("%v", v))
		case "max":
			result = append(result, "-m", fmt.Sprintf("%v", v))
		case "all":
			if v == true {
				result = append(result, "-a")
			}
		case "sort":
			result = append(result, "-s", fmt.Sprintf("%v", v))
		}
	}
	return result
}

func mcpToolsFromSchema() []mcpTool {
	schema := buildSchema()
	var tools []mcpTool
	for _, cmd := range schema.Commands {
		// Add base command as list tool.
		tools = append(tools, mcpTool{
			Name:        cmd.Name + "_list",
			Description: cmd.Description,
			InputSchema: flagsToSchema(cmd.Flags),
		})
		for _, sub := range cmd.Subcommands {
			allFlags := append(cmd.Flags, sub.Flags...)
			tools = append(tools, mcpTool{
				Name:        cmd.Name + "_" + sub.Name,
				Description: sub.Description,
				InputSchema: flagsToSchema(allFlags),
			})
		}
	}
	return tools
}

func flagsToSchema(flags []schemaFlag) map[string]any {
	props := map[string]any{}
	for _, f := range flags {
		t := "string"
		if f.Type == "int" {
			t = "number"
		} else if f.Type == "bool" {
			t = "boolean"
		}
		props[f.Name] = map[string]any{"type": t, "description": f.Description}
	}
	return map[string]any{
		"type":       "object",
		"properties": props,
	}
}
```
